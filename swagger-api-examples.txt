}

{

  "parentId": 1
}
{

  "parentId": 1
}
{

  "parentId": 2

// ===================== PRODUCT =====================
// Example JSON for creating a product
{
  "id": 1,
  "name": "Vòng tay đá phong thủy",
  "description": "Vòng tay đá tự nhiên, mang lại may mắn",
  "category": {
    "id": 3,
    "name": "Vòng tay đá"
  }
}

// Example JSON for updating a product
{
  "name": "Dây chuyền bạc cao cấp",
  "description": "Dây chuyền bạc 925, thiết kế sang trọng",
  "category": {
    "id": 5
  }
}
